\section{\faUsers\ Research Experience}

\datedsubsection{\textbf{World-model-based online RL for Long-Horizon Planning}}{2025 -- Present}
\role{\textbf{Current Research}}{World-Model-Based Online RL}
\begin{itemize}[parsep=0.35ex]
  \item Developing a world-model-based online RL framework for long-horizon planning, extending step-level latent dynamics and policies to trajectory-level world models and action-chunk policies.
  \item Designing trajectory-level policy optimization where action chunks are proposed, evaluated through latent rollouts, and improved using rollout rewards and terminal value estimates under online model updates.
  \item Studying online adaptation challenges in trajectory-level world-model RL, including compounding rollout error, planner-induced distribution shift, and evolving policy targets.

\end{itemize}
