\datedsubsection{\textbf{World Models and Reinforcement Learning}}{2025 -- Present}
\role{\textbf{Doctoral Research}}{William \& Mary}
\begin{itemize}[parsep=0.35ex]
  \item \textbf{WestWorld.} Co-developed a knowledge-encoded trajectory world model for dynamics prediction and model-based control across diverse robotic systems.
  \item \textbf{SWAN.} Developed action-chunk policies and advantage-guided proposal optimization for efficient online planning, using 50\% fewer MPPI refinement iterations across 31 continuous-control tasks. Demonstrated zero-shot sim-to-real transfer to a physical Unitree Go2 for stable velocity tracking.
  \item \textbf{Ongoing: Continual RL.} Investigating humanoid whole-body control through sequential task learning, focusing on skill transfer, sample efficiency, and retention of previously learned skills.
\end{itemize}

\datedsubsection{\textbf{JPMorgan Chase \& Co.}, New York, NY}{06/2024 -- 08/2024; 09/2024 -- 12/2024}
\role{\textbf{Summer Research Associate / Part-time Researcher}}{Leader: Shaohan Hu}
\begin{itemize}[parsep=0.35ex]
  \item Proposed \textbf{ECA} for continual alignment in pretrained vision-language models, combining query mixing, Fisher-guided adapter expansion, and dictionary replay.
  \item Built four topic-shifted image captioning and VQA benchmarks to evaluate adaptation and forgetting in open-ended image-to-text generation.
\end{itemize}

\datedsubsection{\textbf{Continual Learning}}{2023 -- Present}
\role{\textbf{Doctoral Research}}{William \& Mary}
\begin{itemize}[parsep=0.35ex]
  \item \textbf{YoooP.} Developed attentional mean-shift prototype optimization to reduce inter-class interference, enabling class-incremental learning by replaying one prototype per class without storing old images.
  \item \textbf{Hybrid Memory Replay.} Combined checkpoint-based data distillation with complementary real-exemplar selection to mitigate catastrophic forgetting under a limited replay memory budget.
\end{itemize}
